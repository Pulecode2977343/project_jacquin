// STUDENT SCHEDULE REQUEST
// ==========================================
window.openScheduleRequestModal = async function(courseId, courseName) {
    const { value: formValues } = await Swal.fire({
        title: 'Solicitar Horario Adicional',
        html: `
            <div style="text-align:left;">
                <p>Estás solicitando un nuevo horario para: <b>${courseName}</b></p>
                <label style="display:block; margin-top:10px;">Día Preferido</label>
                <select id="req-day" class="swal2-select" style="width:100%;">
                    <option value="Lunes">Lunes</option>
                    <option value="Martes">Martes</option>
                    <option value="Miércoles">Miércoles</option>
                    <option value="Jueves">Jueves</option>
                    <option value="Viernes">Viernes</option>
                    <option value="Sábado">Sábado</option>
                </select>
                <label style="display:block; margin-top:10px;">Hora Preferida</label>
                <input id="req-time" type="time" class="swal2-input" style="width:100%;">
            </div>
        `,
        focusConfirm: false,
        showCancelButton: true,
        confirmButtonText: 'Enviar Solicitud',
        cancelButtonText: 'Cancelar',
        confirmButtonColor: 'var(--color-acento-azul)',
        preConfirm: () => {
            return [
                document.getElementById('req-day').value,
                document.getElementById('req-time').value
            ]
        }
    });

    if (formValues) {
        const [day, time] = formValues;
        if(!day || !time) return;

        const session = ApiService.getSession();
        try {
            const response = await fetch('/jacquin_api/request_schedule.php', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    student_id: session.id_usuario,
                    course_id: courseId,
                    day: day,
                    time: time
                })
            });
            const result = await response.json();
            if(result.success) {
                Swal.fire('Solicitud Enviada', result.message, 'success');
            } else {
                Swal.fire('Error', result.message || 'No se pudo enviar', 'error');
            }
        } catch(e) {
            console.error(e);
            Swal.fire('Error', 'Error de conexión', 'error');
        }
    }
};

window.openStudentTasks = function() {
    let overlay = document.getElementById('student-tasks-overlay');
    if (!overlay) {
        overlay = document.createElement('div');
        overlay.id = 'student-tasks-overlay';
        overlay.style.cssText = "position:fixed; top:0; left:0; width:100%; height:100%; background:rgba(0,0,0,0.9); z-index:10005; display:flex; justify-content:center; align-items:center; opacity:0; transition:opacity 0.3s;";
        overlay.onclick = (e) => { if (e.target === overlay) closeStudentTasks(); };
        document.body.appendChild(overlay);
    }

    overlay.innerHTML = `
        <div style="background:#1e1e1e; width:90%; max-width:800px; height:80vh; border-radius:15px; overflow:hidden; border:1px solid #333; display:flex; flex-direction:column; box-shadow:0 0 50px rgba(0,0,0,0.7);">
            <!-- Header -->
            <div style="background:linear-gradient(135deg, var(--color-principal-azul), #1a5276); padding:20px 30px; display:flex; justify-content:space-between; align-items:center; border-bottom:1px solid rgba(255,255,255,0.1);">
                <div>
                    <h2 style="margin:0; color:white; font-size:1.8rem;"><i class="bi bi-list-check"></i> Mis Tareas</h2>
                    <p style="margin:5px 0 0 0; color:rgba(255,255,255,0.6);">Revisa tus asignaciones pendientes y entregadas</p>
                </div>
                <button onclick="closeStudentTasks()" style="background:none; border:none; color:white; font-size:2rem; cursor:pointer;">&times;</button>
            </div>
            
            <!-- Content -->
            <div id="student-tasks-content" style="flex:1; overflow-y:auto; padding:30px;">
                <div style="text-align:center; color:#666; padding:50px;">
                    <i class="bi bi-arrow-repeat spin"></i> Cargando tareas...
                </div>
            </div>
        </div>
    `;

    overlay.style.display = 'flex';
    setTimeout(() => {
        overlay.style.opacity = '1';
        loadStudentTasks();
    }, 10);
};

window.closeStudentTasks = function() {
    const overlay = document.getElementById('student-tasks-overlay');
    if (overlay) {
        overlay.style.opacity = '0';
        setTimeout(() => overlay.style.display = 'none', 300);
    }
};

window.loadStudentTasks = async function() {
    const container = document.getElementById('student-tasks-content');
    const session = ApiService.getSession();
    
    try {
        const response = await fetch(`/jacquin_api/get_student_tasks.php?student_id=${session.id_usuario}`);
        const result = await response.json();
        
        if(result.success && result.data && result.data.length > 0) {
            container.innerHTML = result.data.map(task => {
                const isSubmitted = task.submission_status === 'submitted' || task.submission_status === 'graded';
                const statusColor = isSubmitted ? '#2ecc71' : '#e74c3c';
                const statusText = isSubmitted ? 'Entregado' : 'Pendiente';
                
                return `
                    <div onclick="openTaskDetail(${task.assignment_id})" 
                         style="background:rgba(255,255,255,0.05); border-radius:10px; padding:20px; margin-bottom:15px; cursor:pointer; border-left:4px solid ${statusColor}; transition:transform 0.2s;">
                        <div style="display:flex; justify-content:space-between; align-items:flex-start;">
                            <div>
                                <h3 style="margin:0 0 5px 0; color:white;">${task.title}</h3>
                                <div style="color:var(--color-acento-azul); font-size:0.9rem; margin-bottom:5px;">${task.course_name}</div>
                                <div style="color:#aaa; font-size:0.85rem;">Prof. ${task.teacher_name}</div>
                            </div>
                            <div style="text-align:right;">
                                <span style="background:${statusColor}; color:white; padding:4px 10px; border-radius:12px; font-size:0.75rem; font-weight:bold;">${statusText}</span>
                                <div style="margin-top:5px; color:#888; font-size:0.8rem;">
                                    Vence: ${new Date(task.due_date).toLocaleDateString()}
                                </div>
                            </div>
                        </div>
                    </div>
                `;
            }).join('');
        } else {
            container.innerHTML = `
                <div style="text-align:center; padding:50px; color:#666;">
                    <i class="bi bi-clipboard-check" style="font-size:3rem; margin-bottom:15px; display:block;"></i>
                    <p>No tienes tareas pendientes.</p>
                </div>
            `;
        }
    } catch(e) {
        container.innerHTML = '<p style="color:red; text-align:center;">Error cargando tareas.</p>';
    }
};

window.openTaskDetail = async function(assignmentId) {
    const session = ApiService.getSession();
    const response = await fetch(`/jacquin_api/get_student_tasks.php?student_id=${session.id_usuario}`);
    const result = await response.json();
    const task = result.data.find(t => t.assignment_id == assignmentId);
    
    if(!task) return;

    Swal.fire({
        title: `Tarea: ${task.title}`,
        html: `
            <div style="text-align:left; font-size:0.95rem;">
                <div style="background:rgba(255,255,255,0.05); padding:15px; border-radius:8px; margin-bottom:15px;">
                    <p style="margin:0 0 10px 0;"><b>Descripción:</b></p>
                    <p style="white-space:pre-wrap; color:#ddd;">${task.description || 'Sin descripción'}</p>
                    ${task.media_url ? `<div style="margin-top:10px;"><a href="${task.media_url}" target="_blank" style="color:var(--color-acento-azul); text-decoration:underline;">Ver Adjunto del Profesor</a></div>` : ''}
                </div>
                
                <hr style="border-color:rgba(255,255,255,0.1);">
                
                <h4 style="margin:15px 0 10px 0; color:var(--color-acento-naranja);">Tu Entrega</h4>
                
                ${task.submission_status === 'graded' ? `
                    <div style="background:rgba(46, 204, 113, 0.1); padding:10px; border-radius:8px; border:1px solid #2ecc71; margin-bottom:15px;">
                        <strong style="color:#2ecc71;">¡Calificado!</strong> Nota: <b>${task.grade}</b>
                    </div>
                ` : ''}

                <label style="display:block; margin-bottom:5px;">Respuesta (Texto):</label>
                <textarea id="sub-text" class="swal2-textarea" style="margin:0 0 15px 0; width:100%;">${task.submission_text || ''}</textarea>
                
                <label style="display:block; margin-bottom:5px;">Archivo Adjunto (Opcional):</label>
                <input type="file" id="sub-file" class="swal2-file" style="margin:0 0 15px 0; width:100%;">
                
                <div id="sub-progress" style="display:none; color:var(--color-acento-azul); font-size:0.9rem; margin-top:10px;">
                    <i class="bi bi-cloud-upload"></i> Subiendo entrega...
                </div>
            </div>
        `,
        width: 600,
        showCancelButton: true,
        confirmButtonText: 'Entregar Tarea',
        cancelButtonText: 'Cerrar',
        confirmButtonColor: 'var(--color-acento-naranja)',
        preConfirm: async () => {
            const text = document.getElementById('sub-text').value;
            const fileInput = document.getElementById('sub-file');
            const file = fileInput.files[0];
            
            document.getElementById('sub-progress').style.display = 'block';
            
            // Build FormData
            const formData = new FormData();
            formData.append('assignment_id', assignmentId);
            formData.append('student_id', session.id_usuario);
            formData.append('text_content', text);
            if(file) {
                formData.append('file', file);
            }

            try {
                const uploadRes = await fetch('/jacquin_api/submit_task.php', {
                    method: 'POST',
                    body: formData
                });
                const uploadResult = await uploadRes.json();
                if(!uploadResult.success) {
                    Swal.showValidationMessage(uploadResult.message);
                }
                return uploadResult;
            } catch(e) {
                Swal.showValidationMessage('Error de conexión');
            }
        }
    }).then((result) => {
        if(result.isConfirmed) {
            Swal.fire('¡Enviado!', 'Tu tarea ha sido entregada exitosamente.', 'success');
            loadStudentTasks(); // Reload
        }
    });

    // Disable inputs if graded
    if (task.submission_status === 'graded') {
        setTimeout(() => {
            const txt = document.getElementById('sub-text');
            const fl = document.getElementById('sub-file');
            if(txt) txt.disabled = true;
            if(fl) fl.disabled = true;
            const confirmBtn = Swal.getConfirmButton();
            if(confirmBtn) confirmBtn.style.display = 'none';
        }, 100);
    }
};
